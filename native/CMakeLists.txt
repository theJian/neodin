cmake_minimum_required(VERSION 3.20)
project(neodin_rpc C)
file(GLOB MPACK_SOURCES CONFIGURE_DEPENDS ../vendor/mpack/*.c)
find_path(UTF8PROC_INCLUDE_DIR utf8proc.h HINTS "${CMAKE_BINARY_DIR}/deps/include" /opt/homebrew/include REQUIRED)
find_path(LIBUV_INCLUDE_DIR uv.h HINTS "${CMAKE_BINARY_DIR}/deps/include" /opt/homebrew/include REQUIRED)
add_library(neodin_rpc STATIC rpc.c fonts.c unicode.c transport.c ${MPACK_SOURCES})
target_include_directories(neodin_rpc PRIVATE ${UTF8PROC_INCLUDE_DIR} ${LIBUV_INCLUDE_DIR})
if(UNIX AND NOT APPLE)
  find_package(Fontconfig REQUIRED)
  target_link_libraries(neodin_rpc PRIVATE Fontconfig::Fontconfig)
endif()
target_include_directories(neodin_rpc PRIVATE ../vendor/mpack)
target_compile_definitions(neodin_rpc PRIVATE MPACK_EXTENSIONS=1)
target_compile_features(neodin_rpc PRIVATE c_std_99)
set_target_properties(neodin_rpc PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" )
foreach(CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(neodin_rpc PROPERTIES ARCHIVE_OUTPUT_DIRECTORY_${CONFIG} "${CMAKE_BINARY_DIR}")
endforeach()
